cmake_minimum_required(VERSION 3.16)
project(LHPublicConsumer LANGUAGES CXX)
find_package(LH 1 CONFIG REQUIRED)
add_executable(public_consumer main.cpp)
target_link_libraries(public_consumer PRIVATE LH::runtime_session_service LH::project_command_coordinator)
get_target_property(publicIncludes LH::runtime_session_service INTERFACE_INCLUDE_DIRECTORIES)
foreach(path IN LISTS publicIncludes)
    if(path MATCHES "[/\\]src([/\\]|$)" OR path MATCHES "D:/Table/LH/include")
        message(FATAL_ERROR "Installed target leaks source includes: ${path}")
    endif()
endforeach()
if(TARGET Qt5::Widgets OR TARGET Qt5::Charts)
    message(FATAL_ERROR "Core consumer unexpectedly imports UI dependencies")
endif()
